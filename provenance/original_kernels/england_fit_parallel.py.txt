"""Parallel CPU fitting of independent members of the unchanged V17 DDQN ensemble.

Same initialization, bootstrap, FQI target, minibatch shuffling, optimizer and
network architecture as primal_dual8_fast.planner.fit_ddqn.  Training members
are independent, hence can run concurrently on separate CPU cores.  Read-only
NumPy arrays use Linux fork's copy-on-write memory; NO nested process pools.
Member checkpoints are input-hash guarded and atomically committed.
"""
from __future__ import annotations
import copy
import hashlib
import json
import multiprocessing as mp
import os
import time
from pathlib import Path
import numpy as np
import torch
from torch import nn
from torch.utils.data import TensorDataset, DataLoader
from primal_dual8_fast import planner

_CONTEXT = None


def _hash_array(h, arr):
    a = np.ascontiguousarray(arr)
    h.update(str(a.dtype).encode())
    h.update(str(a.shape).encode())
    h.update(memoryview(a).cast('B'))


def _input_digest(ds, config, seed, n_actions, budget, base_lambda, lambda_social):
    h = hashlib.sha256(b'V17.1 V17 FQI ensemble member exact input v1\0')
    for a in (ds.states, ds.actions, ds.rewards, ds.next_states, ds.done, ds.next_mask):
        _hash_array(h, a)
    p = Path(planner.__file__)
    h.update(hashlib.sha256(p.read_bytes()).digest())
    h.update(hashlib.sha256(Path(__file__).read_bytes()).digest())
    h.update(json.dumps(dict(config=config, seed=int(seed), n_actions=int(n_actions),
                             base_lambda=base_lambda, lambda_social=lambda_social,
                             budget_target=(None if budget is None else float(budget.target))),
                        sort_keys=True, default=str).encode())
    return h.hexdigest()


def _save_member(directory, mi, digest, state, logs):
    directory.mkdir(parents=True, exist_ok=True)
    weight = directory / f'member_{mi}.pt'
    meta = directory / f'member_{mi}.json'
    tmp = directory / f'member_{mi}.pt.tmp.{os.getpid()}'
    # Atomic weight write then atomic manifest write. An interrupted write isn't reusable.
    torch.save(state, tmp)
    with tmp.open('rb') as f:
        os.fsync(f.fileno())
    sha = hashlib.sha256(tmp.read_bytes()).hexdigest()
    os.replace(tmp, weight)
    obj = {'member': mi, 'fit_input_digest': digest, 'weight_sha256': sha,
           'logs': logs, 'complete': True}
    tmp_meta = directory / f'member_{mi}.json.tmp.{os.getpid()}'
    with tmp_meta.open('w') as f:
        json.dump(obj, f, indent=2, sort_keys=True)
        f.flush()
        os.fsync(f.fileno())
    os.replace(tmp_meta, meta)


def _load_member(directory, mi, digest):
    weight = directory / f'member_{mi}.pt'
    meta = directory / f'member_{mi}.json'
    if not (weight.is_file() and meta.is_file()):
        return None
    obj = json.loads(meta.read_text())
    if (obj.get('fit_input_digest') != digest or obj.get('member') != mi or not obj.get('complete')):
        raise RuntimeError(f'DDQN fit checkpoint input mismatch for member {mi}; use a new output directory')
    if obj.get('weight_sha256') != hashlib.sha256(weight.read_bytes()).hexdigest():
        raise RuntimeError(f'Corrupt DDQN fit checkpoint for member {mi}')
    state = torch.load(weight, map_location='cpu', weights_only=True)
    return state, obj['logs']


def _init(context):
    global _CONTEXT
    _CONTEXT = context
    torch.set_num_threads(1)


def _member_fit(mi):
    c = _CONTEXT
    X, XN = c['X'], c['XN']
    A, D, R, next_mask = c['A'], c['D'], c['R'], c['next_mask']
    cfg, seed, n_actions = c['config'], c['seed'], c['n_actions']
    dev = torch.device('cpu')
    torch.set_num_threads(1)
    print(f'FIT member {mi+1}/{c["members"]} START pid={os.getpid()}', flush=True)
    begin = time.perf_counter()

    tensor_XN = torch.from_numpy(XN).to(dev)
    masks = torch.from_numpy(next_mask).to(dev)
    valid = np.arange(len(A))
    torch.manual_seed(int(seed) + 1009 * mi)
    online = planner.QNetwork(X.shape[1], n_actions).to(dev)
    target = planner.QNetwork(X.shape[1], n_actions).to(dev)
    target.load_state_dict(online.state_dict())
    rng = np.random.default_rng(seed + mi * 7919)
    boot = rng.choice(valid, size=len(valid), replace=True)
    logs = []
    iters = int(cfg['fqi_iterations'])
    for it in range(1, iters + 1):
        online.eval(); target.eval()
        with torch.no_grad():
            qnext=[]
            for start in range(0, len(A), 32768):
                x=tensor_XN[start:start+32768]
                qo=online(x); mt=masks[start:start+32768]
                choice=qo.masked_fill(~mt,-1e12).argmax(dim=1)
                qt=target(x).gather(1,choice[:,None]).squeeze(1)
                qnext.append(qt.cpu().numpy())
            qnext=np.concatenate(qnext)
        Y=(R+(1-D)*qnext).astype(np.float32)
        torch_gen=torch.Generator();torch_gen.manual_seed(seed+mi*100003+it)
        loader=DataLoader(TensorDataset(torch.from_numpy(X[boot]),torch.from_numpy(A[boot]),torch.from_numpy(Y[boot])),
              batch_size=int(cfg['batch_size']),shuffle=True,generator=torch_gen)
        optimizer=torch.optim.AdamW(online.parameters(),lr=float(cfg['learning_rate']),weight_decay=float(cfg['weight_decay']))
        online.train()
        for _ in range(int(cfg['epochs_per_iteration'])):
            for xb,ab,yb in loader:
                xb=xb.to(dev);ab=ab.to(dev);yb=yb.to(dev)
                pred=online(xb).gather(1,ab.long()[:,None]).squeeze(1)
                loss=torch.nn.functional.smooth_l1_loss(pred,yb)
                optimizer.zero_grad(set_to_none=True);loss.backward()
                nn.utils.clip_grad_norm_(online.parameters(),float(cfg['grad_clip_norm']))
                optimizer.step()
        if it%int(cfg['target_sync_every'])==0:
            target.load_state_dict(copy.deepcopy(online.state_dict()))
        with torch.no_grad():
            idx=rng.choice(len(A),size=min(len(A),10000),replace=False)
            xt=torch.from_numpy(X[idx]).to(dev)
            err=online(xt).gather(1,torch.from_numpy(A[idx]).long().to(dev)[:,None]).squeeze(1).cpu().numpy()-Y[idx]
            logs.append({'member':mi,'iteration':it,'held_training_subset_rmse':float(np.sqrt(np.mean(err*err)))})
        if it == 1 or it % 6 == 0 or it == iters:
            print(f'FIT member {mi+1}/{c["members"]} FQI {it}/{iters} '
                  f'elapsed={time.perf_counter()-begin:.1f}s', flush=True)
    state={k:v.cpu().detach().clone() for k,v in online.state_dict().items()}
    if c['checkpoint_dir'] is not None:
        _save_member(c['checkpoint_dir'],mi,c['digest'],state,logs)
    print(f'FIT member {mi+1}/{c["members"]} FINISHED '
          f'elapsed={time.perf_counter()-begin:.1f}s', flush=True)
    return mi, state, logs


def fit_ddqn(ds,n_actions,config,seed,device='cpu',budget=None,base_lambda=0.,lambda_social=None,
             workers=3,checkpoint_dir=None):
    """Compute unchanged V17 fitted-Q ensemble, parallel across independent nets.

    Results are returned in member order (0,1,2) regardless of scheduling.
    When `workers=1`, use the SAME member trainer for equivalence checking.
    Durable per-member checkpoints are keyed by full training dataset contents.
    """
    if device != 'cpu':
        raise ValueError('CPU-only parallel fitting; no forked CUDA contexts')
    members = int(config['ensemble_size'])
    if not 1 <= int(workers) <= 64:
        raise ValueError('workers must be positive and <=64')
    if int(workers)>1 and os.name != 'posix':
        raise RuntimeError('Multi-worker fitting needs Linux fork; use workers=1')
    torch.set_num_threads(1)
    mean=ds.states.mean(axis=0).astype(np.float32)
    sd=np.maximum(ds.states.std(axis=0),1e-6).astype(np.float32)
    X=(ds.states-mean)/sd; XN=(ds.next_states-mean)/sd
    R=ds.rewards
    if lambda_social is not None:
        if budget is None:raise ValueError('Re-pricing reward requires budget')
        R=planner.reprice_training_rewards(R,ds.step,ds.actions,budget,base_lambda,lambda_social)
    scaler={'mean':mean,'sd':sd}
    directory=Path(checkpoint_dir) if checkpoint_dir is not None else None
    digest=_input_digest(ds,config,seed,n_actions,budget,base_lambda,lambda_social)
    output={}
    if directory is not None:
        for mi in range(members):
            result=_load_member(directory,mi,digest)
            if result is not None:
                output[mi]=result
                print(f'FIT member {mi+1}/{members} REUSED verified checkpoint',flush=True)
    pending=[i for i in range(members) if i not in output]
    if pending:
        ctx=dict(X=X,XN=XN,A=ds.actions,D=ds.done,R=R,next_mask=ds.next_mask,
                 config=config,seed=int(seed),n_actions=int(n_actions),members=members,
                 checkpoint_dir=directory,digest=digest)
        if min(int(workers),len(pending))==1:
            _init(ctx)
            for mi in pending:
                _,state,logs=_member_fit(mi)
                output[mi]=(state,logs)
        else:
            with mp.get_context('fork').Pool(processes=min(int(workers),len(pending)),
                                             initializer=_init,initargs=(ctx,)) as pool:
                for mi,state,logs in pool.imap_unordered(_member_fit,pending,chunksize=1):
                    output[mi]=(state,logs)
    if len(output)!=members:
        raise RuntimeError('Incomplete DDQN ensemble')
    nets=[];logs=[]
    for mi in range(members):
        state,member_logs=output[mi]
        net=planner.QNetwork(X.shape[1],n_actions)
        net.load_state_dict(state)
        net.eval()
        nets.append(net)
        logs.extend(member_logs)
    return nets,scaler,logs
